  devOf(s) {
    var d = (s && s.dev) || (this.props && this.props.dev);
    if (d === 'm' || d === 't' || d === 'd') return d;
    if (typeof window !== 'undefined' && window.__NT_AUTO_DEV === true) { var w = window.innerWidth || 1440; return w <= 599 ? 'm' : (w <= 1023 ? 't' : 'd'); }
